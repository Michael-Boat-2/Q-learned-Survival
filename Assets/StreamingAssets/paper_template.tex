\subsection{Tabular Q-learning agent}
\begin{equation}
Q(s,a) \leftarrow (1-\alpha)\,Q(s,a) + \alpha\left[r + \gamma \max_{a' \in \mathcal{A}(s')} Q(s',a')\right]
\end{equation}
\todo{Optimistic initialisation, action masking, terminal vs timeout bootstrapping,
